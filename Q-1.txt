DECLARE
   
   CURSOR c_books IS
      SELECT book_id, title, price
      FROM book;

   
   v_book_id  book.book_id%TYPE;
   v_title    book.title%TYPE;
   v_price    book.price%TYPE;

BEGIN
   
   DBMS_OUTPUT.PUT_LINE(RPAD('-', 50, '-'));
   DBMS_OUTPUT.PUT_LINE(RPAD('BOOK ID', 10) || RPAD('TITLE', 30) || 'PRICE');
   DBMS_OUTPUT.PUT_LINE(RPAD('-', 50, '-'));

  
   OPEN c_books;

  
   LOOP
      
      FETCH c_books INTO v_book_id, v_title, v_price;

      
      EXIT WHEN c_books%NOTFOUND;

      
      DBMS_OUTPUT.PUT_LINE(
         RPAD(TO_CHAR(v_book_id), 10) || 
         RPAD(v_title, 30) || 
         TO_CHAR(v_price, '999,999.00')
      );
   END LOOP;

  
   CLOSE c_books;

EXCEPTION
   WHEN OTHERS THEN
     
      IF c_books%ISOPEN THEN
         CLOSE c_books;
      END IF;
      RAISE;
END;
/
