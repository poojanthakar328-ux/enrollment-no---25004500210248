SET SERVEROUTPUT ON;

DECLARE
      CURSOR c_update_price (p_category book.category%TYPE) IS
      SELECT book_id, title, price
      FROM book
      WHERE UPPER(category) = UPPER(p_category)
      FOR UPDATE OF price;

   v_input_category book.category%TYPE := '&category';
   v_old_price      book.price%TYPE;
   v_new_price      book.price%TYPE;
   v_found          BOOLEAN := FALSE;

BEGIN
   DBMS_OUTPUT.PUT_LINE('Increasing price by 10% for category: ' || v_input_category);
   DBMS_OUTPUT.PUT_LINE(RPAD('-', 70, '-'));
   DBMS_OUTPUT.PUT_LINE(
      RPAD('BOOK ID', 10) || 
      RPAD('TITLE', 30) || 
      RPAD('OLD PRICE', 15) || 
      'NEW PRICE'
   );
   DBMS_OUTPUT.PUT_LINE(RPAD('-', 70, '-'));

   
   FOR rec IN c_update_price(v_input_category) LOOP
      v_found     := TRUE;
      v_old_price := rec.price;
      v_new_price := ROUND(v_old_price * 1.10, 2); -- Increase by 10%

      
      UPDATE book
      SET price = v_new_price
      WHERE CURRENT OF c_update_price;

     
      DBMS_OUTPUT.PUT_LINE(
         RPAD(TO_CHAR(rec.book_id), 10) ||
         RPAD(rec.title, 30) ||
         RPAD(TO_CHAR(v_old_price, '999,990.00'), 15) ||
         TO_CHAR(v_new_price, '999,990.00')
      );
   END LOOP;

   DBMS_OUTPUT.PUT_LINE(RPAD('-', 70, '-'));

   IF NOT v_found THEN
      DBMS_OUTPUT.PUT_LINE('No books found in category: ' || v_input_category);
      ROLLBACK; -- Nothing changed, clean exit
   ELSE
            COMMIT;
      DBMS_OUTPUT.PUT_LINE('Price updates applied and transaction committed successfully.');
   END IF;

EXCEPTION
   WHEN OTHERS THEN
            ROLLBACK;
      DBMS_OUTPUT.PUT_LINE('An error occurred. Changes rolled back.');
      RAISE;
END;
/
